from abc import ABC, abstractmethod


# Base Chef Rules
class PizzaChef(ABC):

    @abstractmethod
    def make_pizza(self):
        pass

    @abstractmethod
    def check_ingredients(self):
        pass


# Cheese pizza chef
class CheeseChef(PizzaChef):

    def make_pizza(self):
        print("Making cheese pizza")

    def check_ingredients(self):
        print("Checking cheese and dough")


# Pepperoni pizza chef
class PepperoniChef(PizzaChef):

    def make_pizza(self):
        print("Making pepperoni pizza")

    def check_ingredients(self):
        print("Checking pepperoni and dough")


# Use chefs
chef1 = CheeseChef()
chef1.make_pizza()

chef2 = PepperoniChef()
chef2.check_ingredients()

What @abstractmethod Actually Means
When you put @abstractmethod on a method, you're saying:
# I am declaring this method EXISTS
# but I am NOT writing what it does
# Every class that inherits from me MUST write their own version of it
