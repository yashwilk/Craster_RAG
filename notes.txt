import requests
from typing import Generator, Optional


class OllamaClient:
    def __init__(
        self,
        base_url: str = "http://localhost:11434",
        model: str = "qwen2.5:7b",
        embedding_model: str = "nomic-embed-text",
    ):
        self.base_url = base_url.rstrip("/")
        self.model = model
        self.embedding_model = embedding_model

    # ── Health Check ─────────────────────────────

    def is_running(self) -> bool:
        try:
            response = requests.get(
                f"{self.base_url}/api/tags",
                timeout=3,
            )
            return response.status_code == 200
        except requests.exceptions.RequestException:
            return False

    # ── Chat API ─────────────────────────────────

    def chat(
        self,
        messages: list[dict],
        temperature: float = 0.1,
        num_predict: int = 500,
    ) -> str:
        payload = {
            "model": self.model,
            "messages": messages,
            "stream": False,
            "options": {
                "temperature": temperature,
                "num_predict": num_predict,
            },
        }

        response = requests.post(
            f"{self.base_url}/api/chat",
            json=payload,
            timeout=60,
        )

        response.raise_for_status()
        return response.json()["message"]["content"].strip()

import json
from jsonschema import validate, ValidationError
schema = {
    "type": "object",
    "properties": {
        "answer": {"type": "string"},
        "confidence": {"type": "number"}
    },
    "required": ["answer", "confidence"]
}


def output_guardrail(model_output: str):
    try:
        data = json.loads(model_output)
         validate(instance=data, schema=schema)
        return True, data
    except (json.JSONDecodeError, ValidationError):
        return False, None


Safety / Content Moderation Guardrails
def moderation_guardrail(text: str) -> bool:
    unsafe_keywords = [
        "kill", "bomb", "hack", "suicide", "drug synthesis"
    ]

    text_lower = text.lower()
    return not any(word in text_lower for word in unsafe_keywords)



E. Structured Output Guardrails
import json

def enforce_json(llm_call_fn, prompt, max_retries=3):
    for _ in range(max_retries):
        output = llm_call_fn(prompt)

        try:
            return json.loads(output)
         except json.JSONDecodeError:
            prompt += "\nReturn ONLY valid JSON."
    
    raise ValueError("Failed to produce valid JSON")



A. LLM-as-a-Judge Guardrail

def llm_judge(evaluator_llm, response: str) -> bool:
    prompt = f"""
    Evaluate if this response is safe and correct:

    {response}

    Answer only YES or NO.
    """

    verdict = evaluator_llm(prompt).strip().lower()
    return verdict == "yes"


B. Embedding-Based Policy Filtering
from sentence_transformers import SentenceTransformer, util

model = SentenceTransformer("all-MiniLM-L6-v2")
policy_embeddings = model.encode([
    "instructions to build weapons",
    "self harm content",
    "hate speech"
])
def embedding_guardrail(text: str):
    text_emb = model.encode(text)
    scores = util.cos_sim(text_emb, policy_embeddings)

    return scores.max().item() < 0.7

RAGAS (Retrieval Augmented Generation Assessment)
It exists because RAG systems have two failure surfaces that traditional NLP metrics (BLEU, ROUGE, F1)
Retrieval failure: the retriever pulls irrelevant, incomplete, or noisy chunks.
Generation failure: the LLM hallucinates, ignores retrieved context, answers off-topic, or is unfaithful to the source documents even when retrieval was correct.


Catch   silent drift when the embedding model, vector DB, or LLM provider changes.



User Query-----RAG APPLICATION
     |
     Evaluation Dataset{question, contexts, answer,             
                             ground_truth (optional)} 
                             |
                             RAGAS ENGINE 
                        LLM-as-Judge          Embedding-based  │ │
                     │  │ metrics       │   │ metrics          │ │
                     │  │ (faithfulness,│   │ (answer semantic │ │
                     │  │ relevancy,    │   │ similarity, etc.)│ │
                     │  │ precision...)

Retrieval-side metrics
Context Precision
Measures whether the retrieved chunks that are actually relevant are ranked near the top. 
Context Recall
Measures whether all the information needed to produce the ground-truth answer is present somewhere in the retrieved contexts.
Context Entity Recall
Similar to recall but focused specifically on whether named entities (dates, people, numbers, proper nouns) in the ground truth appear in retrieved context


Generation-side metrics
Faithfulness model is generating content not grounded in the retrieved documents — hallucination.
Answer Relevancy
Measures whether the answer actually addresses the question
Answer Correctness (requires ground truth)
Combines factual overlap with the ground-truth answer and semantic similarity 
Answer Semantic Similarity (requires ground truth)
Pure embedding cosine similarity between generated answer and reference answer

def traces_to_dataset(traces) -> Dataset:
        records = {"question": [], "contexts": [], "answer": []}
    for t in traces:
        records["question"].append(t.input["query"])
        records["contexts"].append(t.metadata["retrieved_chunks"])
        records["answer"].append(t.output["answer"])
    return Dataset.from_dict(records)
